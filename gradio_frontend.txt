import gradio as gr
from PIL import Image
import numpy as np
import torch
from models.causal_fake_net import CausalFakeNet
from data.pheme_colab import (CLIPExtractor, extract_srl_heuristic)

device = "cuda" if torch.cuda.is_available() else "cpu"

clip = CLIPExtractor(
    device=device,
    num_text_tokens=6,
    num_image_patches=4,
)

model = CausalFakeNet(feat_dim=512, num_gcn_layers=1).to(device)
checkpoint = torch.load('/content/drive/MyDrive/FYP_data/model1.pt', map_location=device)
model.load_state_dict(checkpoint["model_state"])
model.eval()

def analyse_news(news_text, news_image):
    if not news_text or news_text.strip() == "":
        return "⚠️ Please enter news text.", "No text provided to evaluate."

    pil_image = None
    if news_image is not None:
        if isinstance(news_image, np.ndarray):
            pil_image = Image.fromarray(news_image.astype("uint8"))
            print("pil_image from TRUE block", pil_image)
        else:
            pil_image = news_image
            print("pil_image from FALSE block", pil_image)

    text_feats  = clip.extract_text(news_text)
    image_feats = clip.extract_image(pil_image)

    tf  = text_feats.unsqueeze(0).to(device)
    vf  = image_feats.unsqueeze(0).to(device)
    srl = [extract_srl_heuristic(news_text)]

    with torch.no_grad():
        explanations = model.explain(tf, vf, srl)

    explanation = explanations[0]

    if "FAKE" in explanation:
        label = "🔴 FAKE NEWS DETECTED"
    else:
        label = "🟢 REAL NEWS"

    return label, explanation
